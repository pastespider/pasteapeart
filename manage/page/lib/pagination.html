示例代码

$('#test').pagination({
    current: 2,  // 默认为1
    pageSize: 20,  // 默认为10
    total: 1234,  // 默认0
    callback: function (pageno, pagination) {
        $('#output').text('获取第' + pageno + '数据');
    }
})
