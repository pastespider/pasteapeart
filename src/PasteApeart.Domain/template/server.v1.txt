        /// <summary>
        /// 获取
        /// </summary>
        /// <param name="input"></param>
        /// <returns></returns>
        [HttpGet]
        public async Task<PagedResultDto<{{ Name }}ListDto>> Page([FromQuery] InputQuery{{ Name }} input)
        {
            var _query = from a in _dbContext.{{ Name }}
            {%- assign my_index = 0 -%}
            {%- for classProperty in ListPropertys -%}
                {%- if classProperty.class_attributes != null -%}
                    {%- for classAttribute in classProperty.class_attributes -%}
                        {%- if classAttribute.name == "PasteShort" -%}
                            {%- assign my_index = my_index | plus: 1 -%}
                            join b{{my_index}}1 in _dbContext.{{ classAttribute.Args1 }} on a.{{ classProperty.name }} equals b{{my_index}}1.Id into b{{my_index}}2
                            from b{{my_index}}3 in b{{my_index}}2.DefaultIfEmpty()
                        {%- endif -%}
                    {%- endfor -%}
                {%- endif -%}
            {%- endfor -%}

            select new {{ Name }}ListDto
            {
                Id = a.Id,
                {%- assign my_index = 0 -%}
                {%- for classProperty in ListPropertys -%}
                    {%- unless classProperty.build_from_config -%}
                        {{ classProperty.name }} = a.{{ classProperty.name }},
                        {%- if classProperty.class_attributes != null -%}
                            {%- for classAttribute in classProperty.class_attributes -%}
                                {%- if classAttribute.name == "PasteShort" -%}
                                    {%- assign my_index = my_index | plus: 1 -%}
                                    {{ classAttribute.Args2 }} = b{{ my_index }}3 != null ? b{{ my_index }}3.{{ classAttribute.Args3 }} : null,
                                {%- endif -%}
                            {%- endfor -%}
                        {%- endif -%}
                    {%- endunless -%}
                {%- endfor -%}
            };

            _query = _query
            {%- for prop in ListPropertys -%}
                {%- if prop.class_attributes != null -%}
                    {%- for attr in prop.class_attributes -%}
                        {%- if attr.name == "PasteQuery" -%}
                            {%- if prop.property_type == "int" or prop.property_type == "long" -%}
            .WhereIf(input.{{ prop.name }} != 0, x => x.{{ prop.name }} == input.{{ prop.name }})
                            {%- elsif prop.property_type == "string" -%}
            .WhereIf(!String.IsNullOrEmpty(input.{{ prop.name }}), x => x.{{ prop.name }} == input.{{ prop.name }})
                            {%- endif -%}
                        {%- endif -%}
                    {%- endfor -%}
                {%- endif -%}
            {%- endfor -%}
            {%- comment %} 严格按您要求：input.word搜索所有未标记的string字段 {% endcomment -%}
            {%- assign search_fields = ListPropertys | where: "property_type", "string" -%}
            {%- if search_fields.size > 0 -%}
            .WhereIf(!String.IsNullOrEmpty(input.word), x => 
                {%- for field in search_fields -%}
                    {%- unless forloop.first %} || {% endunless -%}
                    x.{{ field.name }}.Contains(input.word)
                {%- endfor -%}
            )
            {%- endif -%}
            ;


            var _pagedto = new PagedResultDto<{{ Name }}ListDto>();
            if (input.page == 1)
            {
                _pagedto.TotalCount = await _query.CountAsync();
            }
            var dataList = await _query
                .OrderByDescending(x => x.Id)
                .Page(input.page, input.size)
                .AsNoTracking()
                .ToListAsync();
            if(dataList==null||dataList.Count == 0){
                return null;
            }
            _pagedto.Items = dataList;
            return _pagedto;
        }
