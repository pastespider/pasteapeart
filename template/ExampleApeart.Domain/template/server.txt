        /// <summary>
        /// 获取
        /// </summary>
        /// <param name="input"></param>
        /// <returns></returns>
        [HttpGet]
        public async Task<PasteResultDto<{{ Name }}ListDto>> Page([FromQuery] InputQuery{{ Name }} input)
        {
            var _query = ReadQueryPage(input,0);
            var _pagedto = new PasteResultDto<{{ Name }}ListDto>();
            if (input.page == 1)
            {
                _pagedto.TotalCount = await _query.CountAsync();
               if(input.request=="one"){
                  _pagedto.BuildModel(new {{ Name }}ListDto(),new InputQuery{{ Name }}()); 
                }
            }
            var dataList = await _query
                .OrderByDescending(x => x.Id)
                .Page(input.page, input.size)
                .AsNoTracking()
                .ToListAsync();
            if(dataList==null||dataList.Count == 0){
               return null;// throw new PasteCodeException("没有查询到数据",204); //由前端处理这个204 由前端处理
            }
            _pagedto.Items = dataList;
            return _pagedto;
        }
